# Code stays precise and secondary to the idea

```ts {2,6-8}
type ResearchSignal = {
  evidence: string
  confidence: number
  implication: string
}

export function frame(signal: ResearchSignal) {
  return `${signal.evidence} → ${signal.implication}`
}
```
